\recept{Kuřecí Tikka Masala}{2 až 12 hodin}{4 porce}{střední}

\textbf{Časy:}
\begin{itemize}
  \item Marinování masa: 2 až 12 hodin
  \item Omáčka: 1 hodina
  \item Chleba: 30 minut
\end{itemize}

\textbf{Příprava masa}
\emph{Čas: 2 až 12 hodin}
\begin{ingredience}
  \item \textbf{500 g} kuřecích prsou
  \item \textbf{1} kelímek bílého jogurtu
  \item \textbf{1 lžíce} nastrouhaného zázvoru
  \item \textbf{1 lžíce} prolisovaného česneku
  \item \textbf{1 lžička} římského mletého kmínu
  \item \textbf{1 lžíce} koření tandoori masala
  \item sůl
\end{ingredience}
\begin{postup}
  \item Maso nakrájet na kostky cca 4×4 cm
  \item Smíchat všechny ingredience s jogurtem v míse
  \item Do směsi naložit maso a nechat marinovat zabalené ve folii
  \item Po marinování naskládat na plech vyložený pečícím papírem a vložit do trouby předehřáté na 250 °C
  \item Péct do měkka asi 10 minut
  \item Hotové maso přidat do omáčky i se šťávou
\end{postup}

\textbf{Omáčka}
\emph{Čas: 1 hodina}
\begin{ingredience}
  \item \textbf{lžíce} oleje (ne olivový)
  \item \textbf{1} velká cibule
  \item \textbf{2} rajčata
  \item \textbf{1 lžíce} nastrouhaného zázvoru
  \item \textbf{1 lžíce} prolisovaného česneku
  \item \textbf{1 lžička} kurkumy
  \item \textbf{1 lžička} římského mletého kmínu
  \item \textbf{1 lžička} indického kari
  \item \textbf{1} kousek celé skořice
  \item \textbf{1} kardamon
  \item \textbf{2} hvězdičky badyánu
  \item \textbf{4} bobkové listy
  \item \textbf{150 ml} smetany ke šlehání (31\%)
  \item \textbf{1} plechovka krájených rajčat
  \item hrst čerstvého koriandru
  \item sůl, cukr
  \item (čerstvá chilli paprička)
\end{ingredience}
\begin{postup}
  \item Na lžíci rozpáleného oleje orestovat najemno nakrájenou cibuly (asi 5 minut)
  \item K cibuly přidat zázvor, česnek, mletý římský kmín, kurkumu a indické kari
  \item Nechat 2 minuty restovat
  \item Přidat na kostičky nakrájená rajčata, skořici, kardamom, badyán, bobkový list, rajčata z plechovky, smetanu a sůl. Může se přidat chilli paprička
  \item Nechat vařit asi 15 minut
  \item Po dochucení omáčky vyndat kardamom, bobkový list, badyán a skořici
\end{postup}

\textbf{Chleba}
\emph{Čas: 30 minut}
\begin{ingredience}
  \item \textbf{250 g} hladké mouky
  \item \textbf{1/2} lžičky kypřícího prášku
  \item \textbf{100 ml} smetany na vaření (12\%)
  \item voda
  \item sůl
  \item černá sezamová semínka
  \item \textbf{1 lžíce} másla
\end{ingredience}
\begin{postup}
  \item Do mísy nasypat mouku s kypřícím ppráškem a trochu osolit
  \item Přilít smetanu
  \item Po troškách přidávat vodu dokud nevznikne kompaktní těsto, nemělo by lepit, aby z něj šly vyválet placky
  \item Těsto rozdělit na 5 částí a každou vyválet na tenkou placku
  \item Placky dát na plech vyložený pečícím papírem, propíchat vidličkou
  \item Dát do trouby na co nejvyšší teplotu a nechat péct cca 3 minuty, dokud se nenafouknou a nezhnědnou
\end{postup}
