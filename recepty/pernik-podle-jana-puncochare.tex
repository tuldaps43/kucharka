\recept{Perník podle Jana Punčocháře}{40 minut}{1 plech}{snadná}

\begin{ingredience}
  \item \textbf{3 ks} vejce
  \item \textbf{180 g} třtinový cukr
  \item \textbf{1 ks} špetka soli
  \item \textbf{1 lžička} jedlá soda
  \item \textbf{40 g} kakao
  \item \textbf{1 ks} prášek do perníku
  \item \textbf{250 ml} mléko
  \item \textbf{120 g} přepuštěné máslo nebo olej
  \item \textbf{350 g} hladká mouka
\end{ingredience}

\begin{postup}
  \item Vejce, cukr a špetku soli vyšlehat do hladké pěny
  \item Přidat lžičku jedlé sody, kakao, prášek do perníku, mléko a máslo a lehce spojit
  \item Vmíchat hladkou mouku aby vzniklo hladké těsto
  \item Můžou se přidat pecičky čokolády
  \item Vylít do vymazaného pekáče
  \item Přidat lžičkami marmeládu
  \item Péct 30 minut na 180 °C horní a spodní ohřev
\end{postup}
